      #cellCountForWordCount(n: number): number {
        // Guard: below the minimum word threshold, default to 3x3 (9) — caller should prevent this.
        if (n < 5) return 9;
        const tier = deriveGridTier(n);
        return tier === "5x5" ? 25 : tier === "4x4" ? 16 : 9;
      }

      #buildBoardForPlayer(wordPool: WordEntry[]): BoardCell[] {
        const cellCount = this.#cellCountForWordCount(wordPool.length);

        // PITFALL 5: copy-per-player (do NOT reuse the shuffled array across players —
        // that would give every player the same board).
        const shuffled = shuffle([...wordPool]);

        const wordCells: BoardCell[] = shuffled.slice(0, cellCount).map((w) => ({
          cellId: nanoid(),
          wordId: w.wordId,
          text: w.text,
          blank: false,
        }));

        const blanksNeeded = Math.max(0, cellCount - wordCells.length);
        const blankCells: BoardCell[] = Array.from({ length: blanksNeeded }, () => ({
          cellId: nanoid(),
          wordId: null,
          text: null,
          blank: true,
        }));

        // Fill-tail then shuffle the combined array so blanks are uniformly distributed
        // across the grid (mathematically equivalent to choosing a random blanks subset).
        return shuffle([...wordCells, ...blankCells]);
