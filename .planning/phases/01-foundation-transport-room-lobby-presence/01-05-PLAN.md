    6. `e2e/reload-identity.spec.ts` (REVIEW — reload preserves sessionStorage identity, no duplicate):
       ```ts
       import { test, expect } from "@playwright/test";

       test("REVIEW: page reload preserves playerId — roster does not duplicate the reloaded player", async ({ browser }) => {
         const ctxA = await browser.newContext();
         const ctxB = await browser.newContext();
         const a = await ctxA.newPage();
         const b = await ctxB.newPage();

         // A creates
         await a.goto("/");
         await a.getByRole("button", { name: "Create a game" }).click();
         await a.getByLabel("Your name").fill("Alice");
         await a.getByRole("button", { name: /Create game/ }).click();
         await a.waitForURL(/\/room\/[A-Z2-9]{6}$/);
         const code = a.url().split("/").pop()!;

         // B joins
         await b.goto(`/join/${code}`);
         await b.getByLabel("Your name").fill("Bob");
         await b.getByRole("button", { name: /Join game/ }).click();
         await b.waitForURL(`**/room/${code}`);
         await expect(a.getByText("Players · 2")).toBeVisible({ timeout: 2000 });

         // Capture B's playerId from sessionStorage before reload
         const bPlayerIdBefore = await b.evaluate((code: string) => {
           const raw = sessionStorage.getItem(`bsbingo_player_${code}`);
           return raw ? JSON.parse(raw).playerId as string : "";
         }, code);
         expect(bPlayerIdBefore).toBeTruthy();

         // Reload B
         await b.reload();
         await b.waitForURL(`**/room/${code}`);

         // After reload, A should STILL see Players · 2 (not 3), and Bob should appear exactly once in A's roster.
         await expect(a.getByText("Players · 2")).toBeVisible({ timeout: 3000 });
         const aBobCount = await a.locator('li', { hasText: 'Bob' }).count();
         expect(aBobCount).toBe(1);

         // B's playerId after reload should equal before reload
         const bPlayerIdAfter = await b.evaluate((code: string) => {
           const raw = sessionStorage.getItem(`bsbingo_player_${code}`);
           return raw ? JSON.parse(raw).playerId as string : "";
         }, code);
         expect(bPlayerIdAfter).toBe(bPlayerIdBefore);

         await ctxA.close();
         await ctxB.close();
       });
       ```

       ```ts
       import { test, expect, request as apiRequest } from "@playwright/test";

       test("CONTRACT: POST /api/rooms returns a valid {code, shareUrl}", async ({ request }) => {
         const res = await request.post("/api/rooms");
         expect(res.status()).toBe(200);
         const body = await res.json() as { code: string; shareUrl: string };
         expect(body.code).toMatch(/^[ABCDEFGHJKMNPQRSTUVWXYZ23456789]{6}$/);
         expect(body.shareUrl).toMatch(new RegExp(`/join/${body.code}$`));
       });

       test("CONTRACT: GET /api/rooms/NOTREAL/exists returns 404", async ({ request }) => {
         const res = await request.get("/api/rooms/NOTREAL/exists");
         expect(res.status()).toBe(404);
       });

       test("CONTRACT + REVIEW: /exists is READ-ONLY — subsequent WS upgrade to the same never-created code returns 404", async ({ request, playwright }) => {
         // Step 1: probe /exists against a never-created code. MUST return 404.
         const existsRes = await request.get("/api/rooms/GHOST01/exists");
         expect(existsRes.status()).toBe(404);

         // Step 2: attempt WS upgrade to the same code. The existence check MUST NOT have initialized the room,
         // so the upgrade MUST ALSO return 404 (NOT 101 + close).
         // Use playwright's request API with Upgrade headers — Playwright will NOT actually complete a WS handshake,
         // but the HTTP response code reveals whether onBeforeConnect returned 404 BEFORE the upgrade.
         const upgradeRes = await request.fetch("/parties/game-room/GHOST01", {
           method: "GET",
           headers: {
             "Upgrade": "websocket",
             "Connection": "Upgrade",
             "Sec-WebSocket-Key": "dGhlIHNhbXBsZSBub25jZQ==",
             "Sec-WebSocket-Version": "13"
           }
         });
         expect(upgradeRes.status()).toBe(404);
       });

       test("CONTRACT + REVIEW: dead-room WS upgrade returns HTTP 404, NOT 101+close", async ({ request }) => {
         const upgradeRes = await request.fetch("/parties/game-room/DEADAB", {
           method: "GET",
           headers: {
             "Upgrade": "websocket",
             "Connection": "Upgrade",
             "Sec-WebSocket-Key": "dGhlIHNhbXBsZSBub25jZQ==",
             "Sec-WebSocket-Version": "13"
           }
         });
         // A true failed upgrade returns 4xx. A "accepted then closed" would be 101 (Switching Protocols).
         expect(upgradeRes.status()).toBe(404);
         expect(upgradeRes.status()).not.toBe(101);
       });
       ```
