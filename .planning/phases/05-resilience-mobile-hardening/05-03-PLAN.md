    **Test 1 — reconnect resume (RESI-03/04)**
    Tag: "reconnect resume"
    Scenario: Create room, join with second player, seed 5 words, start game, verify both see boards. Then take player B offline (setOffline(true)). Assert player B sees reconnecting indicator (text "Reconnecting" visible within 3s). Restore network (setOffline(false)). Assert player B's board reappears within 3s of going back online (board-grid cells visible). Assert the reconnecting indicator is gone.

    Note: The slot-hold window is 45s — Playwright should not wait that long. The test verifies reconnect within the window, not after expiry.

    ```typescript
    test("reconnect resume — board restored after network drop [reconnect]", async ({ browser }) => {
      const ctxA = await browser.newContext();
      const ctxB = await browser.newContext();
      const a = await ctxA.newPage();
      const b = await ctxB.newPage();
      try {
        const code = await createRoom(a, "HostAlice");
        await joinRoom(b, code, "PeerBob");
        await expect(a.getByText("Players · 2")).toBeVisible({ timeout: 5000 });
        await seedWords(a, ["Alpha", "Beta", "Gamma", "Delta", "Epsilon"]);
        await a.getByRole("button", { name: /Start Game/i }).click();
        await expect(a.locator('[data-testid="board-grid"] button').first()).toBeVisible({ timeout: 3000 });
        await expect(b.locator('[data-testid="board-grid"] button').first()).toBeVisible({ timeout: 3000 });

        // Drop B's network
        await ctxB.setOffline(true);
        await expect(b.getByText(/Reconnecting/i)).toBeVisible({ timeout: 4000 });

        // Restore B's network — PartySocket auto-reconnects
        await ctxB.setOffline(false);
        await expect(b.locator('[data-testid="board-grid"] button').first()).toBeVisible({ timeout: 5000 });
        await expect(b.getByText(/Reconnecting/i)).not.toBeVisible({ timeout: 3000 });
      } finally {
        await ctxA.close();
        await ctxB.close();
      }
    });
    ```

    **Test 2 — tab-background resync (RESI-06)**
    Tag: "visibility resync"
    Scenario: Create room, join second player, seed+start game. Player A marks a cell. Player B hides their tab (page.evaluate document.hidden simulation — use Page Visibility API via evaluate). Then player A marks another cell. Player B shows tab again. Assert player B's mark count for A is updated. Use page.evaluate to dispatch visibilitychange:

    ```typescript
    test("tab-background resync — state updates within 1s of tab focus [visibility]", async ({ browser }) => {
      const ctxA = await browser.newContext();
      const ctxB = await browser.newContext();
      const a = await ctxA.newPage();
      const b = await ctxB.newPage();
      try {
        const code = await createRoom(a, "HostAlice");
        await joinRoom(b, code, "PeerBob");
        await expect(a.getByText("Players · 2")).toBeVisible({ timeout: 5000 });
        await seedWords(a, ["Alpha", "Beta", "Gamma", "Delta", "Epsilon"]);
        await a.getByRole("button", { name: /Start Game/i }).click();
        await expect(a.locator('[data-testid="board-grid"] button').first()).toBeVisible({ timeout: 3000 });
        await expect(b.locator('[data-testid="board-grid"] button').first()).toBeVisible({ timeout: 3000 });

        // A marks a cell — B observes the initial count
        await a.locator('[data-testid="board-grid"] button').first().click();
        await expect(b.locator('[data-player-marks]').first()).toBeVisible({ timeout: 2000 });

        // Simulate B's tab going to background: dispatch visibilitychange with state 'hidden'
        await b.evaluate(() => {
          Object.defineProperty(document, "visibilityState", { value: "hidden", configurable: true });
          document.dispatchEvent(new Event("visibilitychange"));
        });

        // A marks another cell while B is "hidden"
        const cells = await a.locator('[data-testid="board-grid"] button').all();
        if (cells.length > 1) await cells[1].click().catch(() => {});

        // Simulate B's tab coming back to foreground: triggers syncRequest
        await b.evaluate(() => {
          Object.defineProperty(document, "visibilityState", { value: "visible", configurable: true });
          document.dispatchEvent(new Event("visibilitychange"));
        });

        // Board should still be visible (syncResponse came in)
        await expect(b.locator('[data-testid="board-grid"] button').first()).toBeVisible({ timeout: 2000 });
      } finally {
        await ctxA.close();
        await ctxB.close();
      }
    });
    ```

    **Test 3 — host failover (RESI-05)**
    Tag: "host-failover"
    Scenario: Create room (A=host), join second player (B). Seed and start game. Close A's connection (setOffline for ctxA). Wait 50s (slot-hold is 45s + buffer) for alarm to fire. Assert B sees "Start" button become visible OR that hostChanged is reflected. This test uses `test.slow()` because it needs to wait past the 45s alarm.

    NOTE: Waiting 50s is necessary to trigger the DO alarm. Set an explicit test timeout of 90s.

    ```typescript
    test("host failover — second player becomes host after slot expires [host-failover]", async ({ browser }) => {
      test.slow(); // extends timeout 3x (default 30s → 90s)
      const ctxA = await browser.newContext();
      const ctxB = await browser.newContext();
      const a = await ctxA.newPage();
      const b = await ctxB.newPage();
      try {
        const code = await createRoom(a, "HostAlice");
        await joinRoom(b, code, "PeerBob");
        await expect(a.getByText("Players · 2")).toBeVisible({ timeout: 5000 });
        await seedWords(a, ["Alpha", "Beta", "Gamma", "Delta", "Epsilon"]);

        // Take host offline (triggers slot-hold countdown)
        await ctxA.setOffline(true);

        // Wait past the 45s slot-hold window for the DO alarm to fire
        await b.waitForTimeout(50_000);

        // B should now be host — see the Start Game button
        await expect(b.getByRole("button", { name: /Start Game/i })).toBeVisible({ timeout: 5000 });
      } finally {
        await ctxA.close();
        await ctxB.close();
      }
    });
    ```
